cmake_minimum_required(VERSION 3.20)
# UI 和外设逻辑可单独在主机测试；产品协议集成测试由 SDK 维护者传入 gagent_core。
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(gagent_example_tests LANGUAGES CXX)
    enable_testing()
endif()
set(EXAMPLE_MAIN "${CMAKE_CURRENT_LIST_DIR}/../examples/p4_iot_basic/main")
add_executable(p4_ui_tests p4_ui_tests.cc)
target_include_directories(p4_ui_tests PRIVATE "${EXAMPLE_MAIN}")
add_test(NAME p4_example_ui COMMAND p4_ui_tests)
add_executable(p4_hardware_tests p4_hardware_tests.cc "${EXAMPLE_MAIN}/hardware.cc")
target_include_directories(p4_hardware_tests PRIVATE fakes "${EXAMPLE_MAIN}")
add_test(NAME p4_example_hardware COMMAND p4_hardware_tests)
foreach(target p4_ui_tests p4_hardware_tests)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
if(TARGET gagent_core)
    add_executable(p4_product_tests p4_product_tests.cc "${EXAMPLE_MAIN}/gizwits_protocol.cc" "${EXAMPLE_MAIN}/gizwits_product.cc")
    target_include_directories(p4_product_tests PRIVATE "${EXAMPLE_MAIN}" product_fakes
        "${EXAMPLE_MAIN}/../gagent-sdk/components/gagent_camera/include")
    target_link_libraries(p4_product_tests PRIVATE gagent_core)
    target_compile_options(p4_product_tests PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
    add_test(NAME p4_example_product COMMAND p4_product_tests)
endif()
