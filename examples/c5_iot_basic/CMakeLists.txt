cmake_minimum_required(VERSION 3.24)
# 一个工程对应一种芯片，防止误用其他目标的静态库。
set(IDF_TARGET esp32c5)
if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/gagent-sdk/gagent-sdk.cmake")
    message(FATAL_ERROR "SDK missing: obtain the complete example including gagent-sdk/.")
endif()
foreach(required main/product_secrets.hpp main/generated/gizwits_protocol.hpp main/generated/gizwits_protocol.cc main/generated/product_identity.hpp)
    if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/${required}")
        message(FATAL_ERROR "Configure this product first: copy main/product_secrets.example.hpp to main/product_secrets.hpp, fill PK/PS, then run python3 gagent-sdk/configure_product.py (or --datapoints path/to/product.local.json).")
    endif()
endforeach()
if("$ENV{IDF_COMPONENT_MANAGER}" STREQUAL "0")
    message(FATAL_ERROR "Enable Component Manager to download pinned dependencies: IDF_COMPONENT_MANAGER=1 idf.py build")
endif()
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/gagent-sdk/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
include("${CMAKE_CURRENT_LIST_DIR}/gagent-sdk/gagent-sdk.cmake")
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(gagent_c5_iot_basic)
gagent_sdk_check()
